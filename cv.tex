\section{Research Interests}
Embodied AI for urban mobility: scalable simulation, navigation foundation models, and
sim-to-real learning that let sidewalk robots navigate safely, robustly, and socially among
pedestrians in real cities.

\section{Research Experience}

\entry{Sep 2025 --\\Present}{%
  \textbf{University of California, Los Angeles}
  \hfill{\small\color{keycol}Los Angeles, CA}\\
  Ph.D. Student \quad{\color{keycol}Advisor: Prof.~\bolei}\\
  Research collaboration with Coco Robotics on sidewalk delivery robots}
\begin{points}
  \item \textbf{Simulation:} Co-first author of SidewalkBench (CoRL 2026 Spotlight; 100 2\,km\,$\times$\,2\,km scenes,
    1{,}235 scenarios, 9 models) and UrbanVerse (ICLR 2026; city-tour videos to simulation).
  \item \textbf{Robot learning:} Led FlowPilot (CoRL 2026; flow matching + human-preference
    alignment), MIMIC (ICRA 2026; multi-scale imitation + corrective behavior expansion),
    and S2E (ICLR 2026; RL fine-tuning of navigation foundation models in simulation).
  \item \textbf{Real-world robots:} Validated these policies on Coco Robotics delivery robots
    over 20+\,km with only a monocular camera and coarse GPS, day and night: 3.4 interventions
    per 100\,m, longest intervention-free run 1.6\,km; FlowPilot cut interventions by 40.0\%.
\end{points}

\entry{Jun 2026 --\\Sep 2026}{%
  \textbf{Coco Robotics}
  \hfill{\small\color{keycol}Los Angeles, CA}\\
  Machine Learning Engineer Intern \quad{\color{keycol}Mentor: \brad{}}}
\begin{points}
  \item \textbf{(1:1) One pilot, one robot:} Improved the driving model with fleet-scale training,
    raising Gaussian-splatting (GS) benchmark success rate from 53.8\% to 91.3\%.
  \item \textbf{(1:N) One pilot, many robots:} Built a scenario-understanding module and human-signal
    interface that hold robots at crosswalks until a pilot approves, letting one pilot supervise many.
\end{points}
